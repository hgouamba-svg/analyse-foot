function confidenceLevel(p){

  const max = Math.max(p.home, p.draw, p.away);

  if(max >= 55) return {label:"Tendance forte", cls:"conf-forte"};

  if(max >= 40) return {label:"Tendance modérée", cls:"conf-moderee"};

  return {label:"Match ouvert", cls:"conf-ouverte"};

}


function noteText(m){

  const p = m.prob;

  const max = Math.max(p.home, p.draw, p.away);

  const favori = max === p.home ? m.home : max === p.away ? m.away : "le nul";

  const gap = Math.abs(p.home - p.away);

  if(max >= 55){

    return `<b>${favori}</b> ressort nettement favori d'après l'estimation (${max.toFixed(1)}%). Écart net avec l'adversaire — signal statistiquement clair, à confronter à l'actualité de l'effectif (blessures, rotation) avant toute décision.`;

  } else if(max >= 40){

    return `Avantage mesuré pour <b>${favori}</b> (${max.toFixed(1)}%), mais l'écart avec les autres issues reste resserré (${gap.toFixed(1)} pts entre les deux équipes). Match à surveiller de près, pas de tendance dominante.`;

  } else {

    return `Aucune issue ne se détache — probabilités proches entre les trois résultats. Match statistiquement <b>imprévisible</b>, la prudence analytique s'impose.`;

  }

}
